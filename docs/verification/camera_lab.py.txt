"""Loopback-only synthetic lab. Does NOT import main/database or start jobs.

Run from repository root: python ../lab_tools/camera_lab.py --port 8767
Requires only stdlib for serving; generate labels separately with the test tool.
"""
import argparse
import json
from http.server import SimpleHTTPRequestHandler, ThreadingHTTPServer
from pathlib import Path
from urllib.parse import parse_qs, unquote, urlsplit

ROOT = Path.cwd().resolve()
PRODUCTS = [
    dict(id=1, code="90001", name="Синтетический товар А", product_type="SYN A", fandom="SYN", image_url="/lab/labels/ean.png"),
    dict(id=2, code="90002", name="Синтетический товар Б", product_type="SYN B", fandom="SYN", image_url="/lab/labels/code128.png"),
    dict(id=3, code="1900010", name="Синтетическое частичное совпадение", product_type="SYN A", fandom="SYN", image_url="/lab/labels/ean.png"),
]


def mapped(p, marking):
    return {"Код": p["code"], "Вид": p["product_type"], "Фандом": p["fandom"], "Название": p["name"],
            "Маркировка": marking, "found_markings": [marking], "table": [], "similar_sticker_match": False}


class Handler(SimpleHTTPRequestHandler):
    def __init__(self, *args, **kwargs):
        super().__init__(*args, directory=str(ROOT), **kwargs)

    def json(self, status, data):
        payload = json.dumps(data, ensure_ascii=False).encode()
        self.send_response(status)
        self.send_header("Content-Type", "application/json; charset=utf-8")
        self.send_header("Cache-Control", "no-store")
        self.end_headers()
        self.wfile.write(payload)

    def do_GET(self):
        parsed = urlsplit(self.path)
        path, q = unquote(parsed.path), parse_qs(parsed.query)
        if path in ('/search', '/baseline-search'):
            file = ROOT / 'static/search.html' if path == '/search' else ROOT.parent / 'search-baseline.html'
            html = file.read_text(encoding='utf-8').replace('<head>', '<head><script src="/lab/search-fixture.js"></script>', 1)
            data = html.encode('utf-8')
            self.send_response(200); self.send_header('Content-Type', 'text/html; charset=utf-8'); self.send_header('Cache-Control', 'no-store'); self.end_headers(); self.wfile.write(data); return
        if path in ('/mark', '/baseline-mark') and q.get('fixture'):
            file = ROOT / 'static/mark.html' if path == '/mark' else ROOT.parent / 'mobile-baseline-mark.html'
            html = file.read_text(encoding='utf-8').replace('</body>', '<script src="/lab/mobile-seed.js"></script></body>')
            data = html.encode('utf-8')
            self.send_response(200); self.send_header('Content-Type', 'text/html; charset=utf-8'); self.send_header('Cache-Control', 'no-store'); self.end_headers(); self.wfile.write(data); return
        if path in ('/static/img/90001.jpg', '/static/img/90002.jpg', '/static/logos/GoJo.jpg'):
            self.path = '/lab/labels/mobile-wide.png' if path.endswith('90002.jpg') or '/logos/' in path else '/lab/labels/mobile-portrait.png'
            return super().do_GET()
        if path == "/api/auth/profile":
            return self.json(200, {"full_name": "SYNTHETIC", "id": 1})
        if path == "/api/mark/status":
            return self.json(200, {"ok": True, "date": "SYNTHETIC", "missing": []})
        if path == "/api/products/filter-options":
            return self.json(200, {"product_types": ["SYN A", "SYN B"], "fandoms": ["SYN"]})
        if path == "/api/products/filtered":
            items = PRODUCTS
            for key in ("code", "product_type", "fandom"):
                if q.get(key):
                    value = q[key][0]
                    items = [p for p in items if value in p[key]] if key == "code" else [p for p in items if p[key] == value]
            offset, limit = int(q.get("offset", [0])[0]), int(q.get("limit", [30])[0])
            return self.json(200, {"items": items[offset:offset + limit], "total": len(items)})
        if path.startswith("/api/products/"):
            product = next((p for p in PRODUCTS if p["code"] == path.rsplit("/", 1)[1]), None)
            return self.json(200 if product else 404, product or {"detail": "Товар не найден"})
        if path.startswith("/api/mark/"):
            raw = path.removeprefix("/api/mark/")
            if raw == "SOURCE-FAIL":
                return self.json(500, {"detail": "SYN unavailable"})
            if raw == "AUTH-EXPIRED":
                return self.json(401, {"detail": "SYN expired"})
            result = []
            if raw in ('MOBILE-DEMO', 'MOBILE-WIDE', 'MOBILE-MISSING', 'MOBILE-MANY'):
                code = '90001' if raw in ('MOBILE-DEMO', 'MOBILE-MANY') else '90002' if raw == 'MOBILE-WIDE' else '90003'
                product = mapped(dict(code=code, name='Вертикальная открытка. Синтетический товар для проверки маркировки', product_type='Открытка', fandom='Тестовый фандом'), 'ktv_WB')
                stickers = [{'sticker': '50000004074', 'row': 1}, {'sticker': '50000007956', 'row': 2}]
                if raw == 'MOBILE-MANY': stickers += [{'sticker': '5000010%04d' % n, 'row': n} for n in range(3, 23)]
                product['table'] = [dict(marking='ktv_WB', platform='WB', id='100000001', bar='2900000000018', stickers=stickers,
                    sticker_to_articul={s['sticker']: code for s in stickers})]
                result = [product]
            if raw in ("90001", "O90001", "2400000900016", "000090001"):
                result = [mapped(PRODUCTS[0], "FSK")]
            if raw in ("2900000000018", "OZNDEMO00001"):
                result = [mapped(PRODUCTS[1], "sia_WB" if raw.isdigit() else "sia_OZ")]
            if raw == "DEMO-COLLISION":
                result = [mapped(PRODUCTS[0], "sia_WB"), mapped(PRODUCTS[1], "kpd_WB")]
            marking = q.get("marking", [""])[0]
            if marking:
                result = [p for p in result if marking in p["found_markings"] or p["Маркировка"] == "FSK"]
            return self.json(200 if result else 404, result or {"detail": "Товар не найден"})
        if path.startswith("/api/"):
            return self.json(404, {"detail": "Unimplemented synthetic API; no upstream"})
        if path == "/mark":
            self.path = "/static/mark.html"
        if path == "/":
            self.path = "/lab/index.html"
        return super().do_GET()

    def do_POST(self):
        # No forwarding or business writes exist in this lab.
        self.json(405, {"detail": "Read-only synthetic lab"})

    def log_message(self, *args):
        pass  # Payloads are not logged.


if __name__ == "__main__":
    parser = argparse.ArgumentParser()
    parser.add_argument("--port", type=int, default=8767)
    args = parser.parse_args()
    print(f"Synthetic lab: http://127.0.0.1:{args.port}/", flush=True)
    ThreadingHTTPServer(("127.0.0.1", args.port), Handler).serve_forever()
